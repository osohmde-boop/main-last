window.startCheckout = async () => { if (!auth.currentUser) { toggleCart(false); return handleAuth(); } if (cart.length === 0) { return Toast.fire({ icon: 'info', title: 'سلة المشتريات فارغة' }); } const itemsPayload = cart.map(it => ({ productId: it.key, quantity: it.qty })); const discountCode = document.getElementById('discount-input')?.value?.trim() || null; toggleCart(false); Swal.fire({ title: 'جاري تحويلك إلى صفحة الدفع...', allowOutsideClick: false, didOpen: () => Swal.showLoading() }); const uid = auth.currentUser.uid; try { const paymentRes = await fetch('https://paddle-backend-self.vercel.app/api/shopier', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ items: itemsPayload, discountCode }) }); const textResponse = await paymentRes.text(); let paymentData; try { paymentData = JSON.parse(textResponse); } catch (err) { Swal.close(); return Swal.fire('خطأ في الاتصال', 'تعذر قراءة بيانات السيرفر بشكل صحيح.', 'error'); } if (!paymentRes.ok || !paymentData.checkoutUrl) { Swal.close(); return Swal.fire('تعذر إتمام الطلب', paymentData.error || 'حدث خطأ، حاول لاحقاً.', 'error'); } const orderData = { userId: uid, customerEmail: auth.currentUser.email, items: cart, total: paymentData.total, whopOrderId: paymentData.orderId, status: 'pending', createdAt: new Date().toISOString() }; await push(ref(db, `orders/${uid}`), orderData); cart = []; updateCartUI(); Swal.close(); // التوجيه المباشر لصفحة Whop window.location.href = paymentData.checkoutUrl; } catch (error) { Swal.close(); Swal.fire('خطأ', 'تعذر الاتصال بالسيرفر. حاول مرة أخرى.', 'error'); } };
